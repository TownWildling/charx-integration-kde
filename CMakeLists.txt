cmake_minimum_required(VERSION 3.22)
project(charx-integration-kde VERSION 0.1.0 LANGUAGES CXX)
find_package(ECM 6.0 REQUIRED NO_MODULE)
list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
include(KDEInstallDirs6)
find_package(Qt6 6.6 REQUIRED COMPONENTS Core Gui)
find_package(KF6CoreAddons 6.0 REQUIRED)
find_package(KF6KIO 6.0 REQUIRED)
find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
set(BUILD_SHARED_LIBS ON)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
set(CMAKE_AUTOMOC ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
configure_file(src/paths.h.in paths.h @ONLY)
kcoreaddons_add_plugin(charxthumbnail SOURCES src/thumbnail.cpp src/assetprocess.cpp src/imagedecoder.cpp INSTALL_NAMESPACE kf6/thumbcreator)
target_include_directories(charxthumbnail PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(charxthumbnail PRIVATE Qt6::Gui KF6::KIOGui KF6::CoreAddons)
install(FILES reader/charx_reader.py DESTINATION ${KDE_INSTALL_LIBEXECDIR}/charx-thumbnailer)
install(FILES data/application-x-charx.xml DESTINATION ${KDE_INSTALL_MIMEDIR})
option(CHARX_INSTALL_PLASMA_HOOK "Install the plugin discovery hook for the current Plasma user" OFF)
if(CHARX_INSTALL_PLASMA_HOOK)
 set(charx_config_home "$ENV{XDG_CONFIG_HOME}")
 if(NOT IS_ABSOLUTE "${charx_config_home}")
  set(charx_config_home "$ENV{HOME}/.config")
 endif()
 string(REPLACE "'" "'\\''" CHARX_PLUGIN_DIR "${KDE_INSTALL_FULL_PLUGINDIR}")
 set(CHARX_PLUGIN_FILE "${CHARX_PLUGIN_DIR}/kf6/thumbcreator/$<TARGET_FILE_NAME:charxthumbnail>")
 configure_file(data/charx-thumbnailer.sh.in charx-thumbnailer.sh.in @ONLY)
 file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/charx-thumbnailer.sh"
      INPUT "${CMAKE_CURRENT_BINARY_DIR}/charx-thumbnailer.sh.in")
 install(FILES "${CMAKE_CURRENT_BINARY_DIR}/charx-thumbnailer.sh"
         DESTINATION "${charx_config_home}/plasma-workspace/env")
endif()
include(CTest)
if(BUILD_TESTING)
 add_test(NAME reader COMMAND ${Python3_EXECUTABLE} -m unittest discover -s tests -v)
 set_tests_properties(reader PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
 add_executable(charx-plugin-smoke tests/plugin_smoke.cpp)
 target_link_libraries(charx-plugin-smoke PRIVATE KF6::KIOGui KF6::CoreAddons Qt6::Gui)
 add_executable(charx-decoder-test tests/decoder_test.cpp src/imagedecoder.cpp)
 target_include_directories(charx-decoder-test PRIVATE src)
 target_link_libraries(charx-decoder-test PRIVATE Qt6::Gui)
 add_test(NAME decoder COMMAND charx-decoder-test)
endif()
